const loadScript = (url) => new Promise((resolve, reject) => {
    if (document.querySelector(`script[src="${url}"]`)) return resolve();
    const s = document.createElement("script");
    s.src = url;
    s.onload = resolve;
    s.onerror = reject;

});

    });

    document.addEventListener("pointerlockchange", () => {
        const isLocked = (document.pointerLockElement === canvas);
        if (mouseBtn) {
            if (isLocked) {
                mouseBtn.innerText = "Unlock Mouse (Esc)";
                mouseBtn.classList.add("active-lock");
            } else {
                mouseBtn.innerText = "Lock Mouse";
                mouseBtn.classList.remove("active-lock");
                // open menu to pause if unlocked via esc
                if (mouseLocked) {
                    sendKey(27, true);
                    setTimeout(() => sendKey(27, false), 80);
                }
            }
        }
        mouseLocked = isLocked;
    }, { signal });

    // fullscreen handling with keyboard lock
    const fsBtn = document.getElementById("doom_fs_btn");
    const fsTarget = document.getElementById("doom_fullscreen_target") || canvas;

    const toggleFullscreen = async () => {
        focusGame();
        const isFull = !!(document.fullscreenElement || document.webkitFullscreenElement);
        try {
            if (!isFull) {
                if (fsTarget.requestFullscreen) {
                    await fsTarget.requestFullscreen();
                } else if (fsTarget.webkitRequestFullscreen) {
                    await fsTarget.webkitRequestFullscreen();
                }
                if (navigator.keyboard && navigator.keyboard.lock) {
                    try {
                        await navigator.keyboard.lock(["Escape", "KeyW", "KeyA", "KeyS", "KeyD", "Tab"]);
                    } catch (_) {}
                }
            } else {
                if (navigator.keyboard && navigator.keyboard.unlock) {
                    try { navigator.keyboard.unlock(); } catch (_) {}
                }
                if (document.exitFullscreen) {
                    await document.exitFullscreen();
                } else if (document.webkitExitFullscreen) {
                    await document.webkitExitFullscreen();
                }
            }
        } catch (err) {
            console.warn("Fullscreen toggle warning:", err);
        }
    };

    fsBtn?.addEventListener("click", toggleFullscreen);

    const onFullscreenChange = () => {
        const isFull = !!(document.fullscreenElement || document.webkitFullscreenElement);
        if (fsBtn) {
            fsBtn.innerText = isFull ? "Exit fullscreen" : "Fullscreen";
            if (isFull) {
                fsBtn.classList.add("active-lock");
                if (navigator.keyboard && navigator.keyboard.lock) {
                    navigator.keyboard.lock(["Escape", "KeyW", "KeyA", "KeyS", "KeyD", "Tab"]).catch(() => {});
                }
            } else {
                fsBtn.classList.remove("active-lock");
                if (navigator.keyboard && navigator.keyboard.unlock) {
                    try { navigator.keyboard.unlock(); } catch (_) {}
                }
            }
        }
        focusGame();
    };

    document.addEventListener("fullscreenchange", onFullscreenChange, { signal });
    document.addEventListener("webkitfullscreenchange", onFullscreenChange, { signal });

    // boot js-dos and run DOOM
    try {
        if (!window.Dos) {
            await loadScript("https://js-dos.com/6.22/current/js-dos.js");
        }

        window.Dos(canvas, {
            wdosboxUrl: "https://js-dos.com/6.22/current/wdosbox.js",
            autolock: false
        }).ready((fs, main) => {
            statusText.innerText = "Extracting DOOM shareware data...";

            fs.extract("https://js-dos.com/cdn/upload/DOOM-@evilution.zip").then(() => {
                statusText.style.display = "none";
                focusGame();

                main(["-c", "cd DOOM", "-c", "DOOM.EXE"]).then((ci) => {
                    window.doomCi = ci;
                });
            });
        });

        // back button to return to candidate selection
        document.getElementById("doom_back_btn")?.addEventListener("click", (evt) => {
            unlockPointer();
            if (document.fullscreenElement || document.webkitFullscreenElement) {
                try {
                    if (document.exitFullscreen) document.exitFullscreen();
                    else if (document.webkitExitFullscreen) document.webkitExitFullscreen();
                } catch (_) {}
            }
            if (window.doomCi?.exit) {
                try { window.doomCi.exit(); } catch (_) {}
            }
            if (keyCtrl) {
                keyCtrl.abort();
                keyCtrl = null;
            }
            window.stopSpacebar = false;
            e.code2Loaded = false;
            e.doom = false;
            $("#tct_doom_styles").remove();
            $("#bottomBar").show();
            $("#featured-mods-area").show();
            corrr = "\n              <h2>The Campaign Trail</h2>\n            ";
            applyTheme(theme);
            window.candSel(evt);
        });

    } catch (err) {
        statusText.style.color = "red";
        statusText.innerText = `Boot failed: ${err.message}`;
    }
};
